      currentSlide = (index + totalSlides) % totalSlides;
      for (let i = 0; i < totalSlides; i++) {
        const slideEl = document.getElementById('banner-slide-' + i);
        const dotEl = document.getElementById('slider-dot-' + i);
        if (slideEl && dotEl) {
          if (i === currentSlide) {

          }
        }
      }
